cmake_minimum_required(VERSION 3.21)
project(MelvorCompanion VERSION 0.1.2 LANGUAGES C CXX)
# GitHub "owner/repo" whose releases the optional update check compares against this version.
# An empty value hides the menu item.
set(MELVOR_COMPANION_UPDATE_REPO "vani5h/melvor-companion" CACHE STRING "GitHub owner/repo for the update check")
configure_file(src/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/version.h)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/W4 /permissive- /Zc:__cplusplus /utf-8)
endif()
add_library(detection STATIC src/melvor.cpp src/process.cpp src/windows.cpp src/visibility.cpp src/audio.cpp)
target_include_directories(detection PUBLIC src)
target_link_libraries(detection PRIVATE user32 advapi32 ole32 uuid)
file(GLOB BROTLI_COMMON CONFIGURE_DEPENDS third_party/brotli/c/common/*.c)
file(GLOB BROTLI_DEC CONFIGURE_DEPENDS third_party/brotli/c/dec/*.c)
add_library(brotli_decoder STATIC ${BROTLI_COMMON} ${BROTLI_DEC})
target_include_directories(brotli_decoder PUBLIC third_party/brotli/c/include)
target_compile_definitions(brotli_decoder PRIVATE BROTLI_STATIC)
add_library(game_activity STATIC src/game_activity.cpp)
target_include_directories(game_activity PUBLIC src)
target_link_libraries(game_activity PRIVATE brotli_decoder shell32 ole32)
add_library(tray STATIC src/death_panel.cpp src/tray.cpp src/log.cpp src/activity.cpp src/settings.cpp src/ui.cpp src/goals.cpp src/summary.cpp src/update.cpp src/taskbar_probe.cpp src/taskbar_widget.cpp)
target_include_directories(tray PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(tray PUBLIC game_activity)
target_link_libraries(tray PUBLIC detection PRIVATE user32 shell32 ole32 comctl32 gdi32 gdiplus uxtheme dwmapi winhttp oleaut32)
add_executable(melvor_companion WIN32 src/main.cpp src/icons.rc src/companion.manifest)
set_target_properties(melvor_companion PROPERTIES OUTPUT_NAME MelvorCompanion)
target_link_libraries(melvor_companion PRIVATE tray)
# icons.rc includes the generated version.h for the VERSIONINFO resource.
target_include_directories(melvor_companion PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

install(TARGETS melvor_companion RUNTIME DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs/images DESTINATION docs)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "MelvorCompanion-${PROJECT_VERSION}-${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_VS_PLATFORM_NAME STREQUAL "x64")
    set(CPACK_PACKAGE_FILE_NAME "MelvorCompanion-${PROJECT_VERSION}-windows-x64")
endif()
set(CPACK_PACKAGE_CHECKSUM SHA256)
include(CPack)

# Development tests and tools are optional and not needed to build the app.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt")
    enable_testing()
    add_subdirectory(tests)
endif()
